```javascript
class MatchEngine {
    constructor(homeTeam, awayTeam) {
        this.homeTeam = homeTeam;
        this.awayTeam = awayTeam;

        this.pressure = 0;

    // ALGORITHM: Calculate match outcome based on:
    calculateMatchOutcome() {
        /*
        1. Team Strength Calculation:
           - Average player ratings (weighted by position importance)
           - Formation effectiveness matrix
           - Tactical style matchup
           - Home advantage (12% boost)
           
        2. Dynamic Match Flow:
           - Every 30 seconds game time:
             * Calculate possession based on midfield strength
             * Determine attacking opportunity (possession * attacking strength)
             * Calculate shot quality (0-100)
             * Apply goalkeeper save probability
             * Factor in defender positioning
             
        3. Event Generation:
           - Goal probability: (Shot Quality - GK Rating) / 100
           - Assist attribution: 70% chance if goal scored
           - Card probability based on:
             * Tackle intensity (tactical setting)
             * Referee strictness (random 1-10)
             * Player aggression attribute
             * Match importance multiplier
        */
        
        // Possession calculation
        const homeMidStrength = this.calculateMidfieldStrength(this.homeTeam);
        const awayMidStrength = this.calculateMidfieldStrength(this.awayTeam);
        this.possession = (homeMidStrength / (homeMidStrength + awayMidStrength)) * 100;
        
        // Shot generation rate (shots per minute)
        const homeShotRate = (this.possession / 100) * 0.3 * this.homeTeam.attackingStrength;
        const awayShotRate = ((100 - this.possession) / 100) * 0.3 * this.awayTeam.attackingStrength;
        
        // Goal conversion rate
        const homeConversion = this.calculateConversionRate(this.homeTeam, this.awayTeam);
        const awayConversion = this.calculateConversionRate(this.awayTeam, this.homeTeam);
    }
    
    calculateMidfieldStrength(team) {
        // Sum of midfielders' passing, vision, stamina
        return team.midfielders.reduce((sum, player) => {
            return sum + (player.passing * 0.4 + player.vision * 0.3 + player.stamina * 0.3);
        }, 0) / team.midfielders.length;
    }
    
    calculateConversionRate(attackingTeam, defendingTeam) {
        const attackPower = attackingTeam.forwards.reduce((sum, p) => sum + p.finishing, 0);
        const defensePower = defendingTeam.defenders.reduce((sum, p) => sum + p.defending, 0);
        const gkPower = defendingTeam.goalkeeper.handling * 1.5;
        
        return (attackPower / (attackPower + defensePower + gkPower)) * 0.15; // 15% max conversion
    }
}
```

```javascript
                return leg1Away > leg2Away ? team2 : team1;
            } else {
                // Extra time and penalties
                return this.simulatePenalties(team1, team2);
            }
        }
        
        return team1Aggregate > team2Aggregate ? team1 : team2;
    }
}
```
